Reaktionsspiel

In diesem Projekt baust du ein einfaches Reaktionsspiel mit Display, RGB-LED, Summer und Taster. Auf dem Display erscheint zuerst ein Startbildschirm. Während der Wartezeit leuchtet die LED rot. Sobald du drücken darfst, wird die LED grün und der Summer macht einen kurzen Ton. Wenn du zu früh drückst, ist die Runde verloren.

Funktionsweise

  • Auf dem Display steht zuerst Start.
  • Mit einem Tastendruck beginnt das Spiel.
  • Während der Wartezeit leuchtet die LED rot.
  • Das Programm wartet 2 bis 5 Sekunden.
  • Wenn du zu früh drückst, steht Zu frueh! auf dem Display.
  • Wenn du nicht zu früh drückst, erscheint irgendwann JETZT! auf dem Display.
  • Gleichzeitig wird die LED grün und der Summer piept kurz.
  • Wenn du nun drückst, wird deine Reaktionszeit angezeigt.
  • Danach wird die Zeit gestoppt.
  • Erst mit einem neuen Tastendruck startet das Spiel neu.

Quellcode

Datei main.py

from machine import Pin, I2C, PWM
import ssd1306
import time
import urandom

# Taster und Display vorbereiten.
button = Pin(1, Pin.IN, Pin.PULL_UP)
i2c = I2C(1, scl=Pin(3), sda=Pin(2))
oled = ssd1306.SSD1306_I2C(128, 32, i2c)

# LED und Summer vorbereiten.
red = PWM(Pin(9))
green = PWM(Pin(7))
buzzer = PWM(Pin(16))

red.freq(1000)
green.freq(1000)

buzzer.freq(1000)
buzzer.duty_u16(0)

def zeige(text1, text2="", text3=""):
    # Schreibt bis zu drei Zeilen auf das Display.
    oled.fill(0)
    oled.text(text1, 0, 0)
    oled.text(text2, 0, 12)
    oled.text(text3, 0, 24)
    oled.show()

def taste_gedrueckt():
    # Der Taster ist mit Pull-Up verbunden: 0 bedeutet gedrueckt.
    return button.value() == 0

def led_aus():
    # Schaltet die RGB-LED aus.
    red.duty_u16(0)
    green.duty_u16(0)

def led_rot():
    # Zeigt: Jetzt warten.
    red.duty_u16(65535)
    green.duty_u16(0)

def led_gruen():
    # Zeigt: Jetzt druecken.
    red.duty_u16(0)
    green.duty_u16(65535)

def warte_auf_taste():
    # Erst warten, dann entprellen, dann auf das Loslassen warten.
    while not taste_gedrueckt():
        time.sleep(0.01)

    time.sleep(0.2)

    while taste_gedrueckt():
        time.sleep(0.01)

def warte_auf_neue_runde():
    # Wartet, bis die Taste losgelassen und neu gedrueckt wurde.
    while taste_gedrueckt():
        time.sleep(0.01)

    warte_auf_taste()

zeige("Reaktionsspiel", "Start mit", "Taste")
led_aus()
warte_auf_taste()

while True:
    # Neue Runde starten.
    zeige("Warte...")
    led_rot()
    zu_frueh = False
    # Zufallszahl zwischen 2 und 5 Sekunden.
    wartezeit = 2 + urandom.getrandbits(2)

    # Waehlt man zu frueh, ist die Runde verloren.
    for _ in range(wartezeit * 100):
        if taste_gedrueckt():
            zu_frueh = True
            break
        time.sleep(0.01)

    if zu_frueh:
        zeige("Zu frueh!", "", "Taste = neu")
        led_rot()
        warte_auf_neue_runde()
        continue

    zeige("JETZT!")
    led_gruen()
    # Kurzen Signalton abspielen.
    buzzer.duty_u16(20000)
    start = time.ticks_ms()

    # Jetzt wird die Reaktionszeit gemessen.
    while not taste_gedrueckt():
        if time.ticks_diff(time.ticks_ms(), start) > 200:
            buzzer.duty_u16(0)
        time.sleep(0.01)

    reaktionszeit = time.ticks_diff(time.ticks_ms(), start)
    buzzer.duty_u16(0)
    led_aus()
    # Gemessene Zeit anzeigen.
    zeige("Zeit:", str(reaktionszeit) + " ms", "Taste = neu")
    warte_auf_neue_runde()
