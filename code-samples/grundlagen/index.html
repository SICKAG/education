Grundlagen

Auf dieser Seite lernst du die Bauteile einzeln anzusteuern. Kopiere immer nur ein Beispiel gleichzeitig in die Datei main.py.

Allgemeine Infos

Anzeige print("[Beliebiger Text]") ermöglicht es dir, Text in Thonny anzeigen zu lassen. Dieser wird in der Kommandozeile (Fenster unten) dargestellt.

Endlosschleife Damit das Programm endlos läuft, wird eine Schleife benötigt. while True: bewirkt, dass der eingerückte Inhalt immer wieder ausgeführt wird. Wenn du ein Programm stoppen willst, klicke in Thonny auf den roten Stopp-Knopf.

Kommentar Kommentare sind ebenfalls möglich und werden zur Erläuterung verwendet. Diese werden durch # erstellt.

1. Weiße LED

Die weiße LED ist an Pin 15 angeschlossen. Sie soll 2 Sekunden leuchten und danach wieder ausgehen. Nach led.off() ist das Programm zu Ende.

from machine import Pin
import time

led = Pin(15, Pin.OUT)

# LED einschalten.
led.on()
# 2 Sekunden warten.
time.sleep(2)
# LED wieder ausschalten.
led.off()

Teste auch: Ändere die 2 zu 1 oder 5.

2. Photodiode

Die Photodiode misst die Lichtintensität. Je heller es ist, desto höher ist der Wert. Dieser soll in der Kommandozeile ausgegeben werden.

from machine import ADC, Pin
import time

sensor = ADC(Pin(28))

while True:
    # Aktuellen Lichtwert lesen.
    wert = sensor.read_u16()
    # Wert in der Kommandozeile ausgeben.
    print(wert)
    time.sleep(0.1)

Teste auch: Halte deine Hand vor den Sensor oder leuchte mit einer Lampe darauf.

3. RGB-LED

Die RGB‑LED ist eine farbige LED. In ihr stecken eigentlich drei LEDs:

  • R = Rot
  • G = Grün
  • B = Blau

Durch das Mischen dieser drei Farben können viele andere Farben entstehen.

Die Farben werden mit PWM gesteuert. Dabei benutzt der Mikrocontroller Zahlen von 0 bis 65535:

  • 0 → LED aus
  • 65535 → LED ganz hell
  • Zahlen dazwischen → LED dazwischen

Du musst noch nicht verstehen, wie PWM technisch funktioniert. Falls es dich interessiert, schau auf der Hardware-Seite vorbei. Wichtig ist erstmal nur: Große Zahl = hell, kleine Zahl = dunkel.

from machine import Pin, PWM

r = PWM(Pin(9))
g = PWM(Pin(7))
b = PWM(Pin(8))

# Alle drei Farben vorbereiten.
r.freq(1000)
g.freq(1000)
b.freq(1000)

# Beispiele:
# Rot   = 65535, 0, 0
# Gruen = 0, 65535, 0
# Blau  = 0, 0, 65535
# Aus   = 0, 0, 0
rot = 65535
gruen = 0
blau = 0

r.duty_u16(rot)
g.duty_u16(gruen)
b.duty_u16(blau)

Teste auch: Lass Rot und Grün gleichzeitig an, um Gelb zu erhalten, oder lasse alle drei Farben leuchten.

4. Taster

Der Taster funktioniert wie ein Knopf. Er ist an Pin 1 angeschlossen und soll beim Drücken einen Text in der Kommandozeile ausgeben. Wichtig: Beim IoT-Sensor bedeutet 0, dass der Taster gedrückt ist.

from machine import Pin
import time

button = Pin(1, Pin.IN, Pin.PULL_UP)

while True:
    # Bei Pull-Up bedeutet 0: Taste ist gedrueckt.
    if button.value() == 0:
        print("Button gedrueckt")
        # Kurze Pause gegen mehrfaches Ausloesen.
        time.sleep(0.2)

time.sleep(0.2) sorgt dafür, dass durch das sogenannte Prellen des Tasters der Text nicht mehrmals auf einmal ausgegeben wird.

5. Buzzer

Der Buzzer ist ein kleiner Summer. In diesem Beispiel erzeugt er für 1 Sekunde einen Ton. freq(1000) bestimmt die Tonhöhe (Frequenz). Die Zahl bei duty_u16(5000) bestimmt die Lautstärke.

from machine import Pin, PWM
import time

buzzer = PWM(Pin(16))

# Tonhoehe einstellen.
buzzer.freq(1000)
# Ton einschalten.
buzzer.duty_u16(30000)
# 1 Sekunde warten.
time.sleep(1)
# Ton wieder ausschalten.
buzzer.duty_u16(0)

Teste auch: Andere Tonhöhen und Lautstärken.

6. Display

Auf dem Display kann Text angezeigt werden. Voraussetzung: Die Datei ssd1306.py ist auf dem Pico. Wenn sie noch fehlt, klappe den Block auf und kopiere sie.

ssd1306.py Datei (Display Treiber) 
# MicroPython SSD1306 OLED driver, I2C and SPI interfaces

from micropython import const
import framebuf


# register definitions
SET_CONTRAST = const(0x81)
SET_ENTIRE_ON = const(0xA4)
SET_NORM_INV = const(0xA6)
SET_DISP = const(0xAE)
SET_MEM_ADDR = const(0x20)
SET_COL_ADDR = const(0x21)
SET_PAGE_ADDR = const(0x22)
SET_DISP_START_LINE = const(0x40)
SET_SEG_REMAP = const(0xA0)
SET_MUX_RATIO = const(0xA8)
SET_COM_OUT_DIR = const(0xC0)
SET_DISP_OFFSET = const(0xD3)
SET_COM_PIN_CFG = const(0xDA)
SET_DISP_CLK_DIV = const(0xD5)
SET_PRECHARGE = const(0xD9)
SET_VCOM_DESEL = const(0xDB)
SET_CHARGE_PUMP = const(0x8D)

class SSD1306(framebuf.FrameBuffer):
    def __init__(self, width, height, external_vcc):
        self.width = width
        self.height = height
        self.external_vcc = external_vcc
        self.pages = self.height // 8
        self.buffer = bytearray(self.pages * self.width)
        super().__init__(self.buffer, self.width, self.height, framebuf.MONO_VLSB)
        self.init_display()

    def init_display(self):
        for cmd in (
            SET_DISP | 0x00,
            SET_MEM_ADDR,
            0x00,
            SET_DISP_START_LINE | 0x00,
            SET_SEG_REMAP | 0x01,
            SET_MUX_RATIO,
            self.height - 1,
            SET_COM_OUT_DIR | 0x08,
            SET_DISP_OFFSET,
            0x00,
            SET_COM_PIN_CFG,
            0x02 if self.width > 2 * self.height else 0x12,
            SET_DISP_CLK_DIV,
            0x80,
            SET_PRECHARGE,
            0x22 if self.external_vcc else 0xF1,
            SET_VCOM_DESEL,
            0x30,
            SET_CONTRAST,
            0xFF,
            SET_ENTIRE_ON,
            SET_NORM_INV,
            SET_CHARGE_PUMP,
            0x10 if self.external_vcc else 0x14,
            SET_DISP | 0x01,
        ):
            self.write_cmd(cmd)
        self.fill(0)
        self.show()

    def poweroff(self):
        self.write_cmd(SET_DISP | 0x00)

    def poweron(self):
        self.write_cmd(SET_DISP | 0x01)

    def contrast(self, contrast):
        self.write_cmd(SET_CONTRAST)
        self.write_cmd(contrast)

    def invert(self, invert):
        self.write_cmd(SET_NORM_INV | (invert & 1))

    def show(self):
        x0 = 0
        x1 = self.width - 1
        if self.width == 64:
            x0 += 32
            x1 += 32
        self.write_cmd(SET_COL_ADDR)
        self.write_cmd(x0)
        self.write_cmd(x1)
        self.write_cmd(SET_PAGE_ADDR)
        self.write_cmd(0)
        self.write_cmd(self.pages - 1)
        self.write_data(self.buffer)

class SSD1306_I2C(SSD1306):
    def __init__(self, width, height, i2c, addr=0x3C, external_vcc=False):
        self.i2c = i2c
        self.addr = addr
        self.temp = bytearray(2)
        self.write_list = [b"\x40", None]
        super().__init__(width, height, external_vcc)

    def write_cmd(self, cmd):
        self.temp[0] = 0x80
        self.temp[1] = cmd
        self.i2c.writeto(self.addr, self.temp)

    def write_data(self, buf):
        self.write_list[1] = buf
        self.i2c.writevto(self.addr, self.write_list)

class SSD1306_SPI(SSD1306):
    def __init__(self, width, height, spi, dc, res, cs, external_vcc=False):
        self.rate = 10 * 1024 * 1024
        dc.init(dc.OUT, value=0)
        res.init(res.OUT, value=0)
        cs.init(cs.OUT, value=1)
        self.spi = spi
        self.dc = dc
        self.res = res
        self.cs = cs
        import time

        self.res(1)
        time.sleep_ms(1)
        self.res(0)
        time.sleep_ms(10)
        self.res(1)
        super().__init__(width, height, external_vcc)

    def write_cmd(self, cmd):
        self.spi.init(baudrate=self.rate, polarity=0, phase=0)
        self.cs(1)
        self.dc(0)
        self.cs(0)
        self.spi.write(bytearray([cmd]))
        self.cs(1)

    def write_data(self, buf):
        self.spi.init(baudrate=self.rate, polarity=0, phase=0)
        self.cs(1)
        self.dc(1)
        self.cs(0)
        self.spi.write(buf)
        self.cs(1)

from machine import Pin, I2C
import ssd1306
import time

# Verbindung zum Display herstellen.
i2c = I2C(1, scl=Pin(3), sda=Pin(2))
oled = ssd1306.SSD1306_I2C(128, 32, i2c)

# Altes Bild loeschen und neuen Text zeichnen.
oled.fill(0)
oled.text("Hallo!", 0, 0)
oled.text("Display Test", 0, 16)
# Inhalt sichtbar machen.
oled.show()

time.sleep(2)
